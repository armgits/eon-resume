    {coursework project}
    {Mar 2024 - Apr 2024}

  \begin{project}
    {Justbot Manipulator}
    {coursework project}
    {Nov 2022 - Dec 2022}

    \item Simulated kinematics of a 7 DOF articulated serial manipulator designed in Solidworks
          using Gazebo and ROS.

    \item Written an inverse kinematics solver in Python that publishes joint
          angles for a given end-effector displacement.

  \end{project}

  \begin{project}
    {Mint Flight Controller Board}
    {personal project}
    {Oct 2020 - May 2021}

    \item Designed a miniature flight controller board for quadcopter drones
          in KiCAD based on STM32H7 microcontroller.

    \item The resulting 4-layer PCB sizes 80 percent smaller than Pixhawk 4
