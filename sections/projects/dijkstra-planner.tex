\begin{project}
  {Dijkstra Planner for Point Robot}
  {C++, Path Planning, Graph Search and 2D Geometry}
  {Mar 2024 - Apr 2024}
  {https://github.com/armgits/project2-dijkstra}

  \item Implemented a Dijkstra planner in C++ using STL to navigate a holonomic point robot through
        a 2D obstacle map.

  \item Developed a custom 2D visualization using OpenGL to demonstrate graph search and
        backtracking.

  \item Leveraged multi-threading to simultaneously visualize and perform the graph search.

\end{project}
